10.2. Konfigurasi Connection Pool & Adaptabilitas Lingkungan Dinamis

Lokasi Berkas: backend/src/config/database.js

Konfigurasi pool mendeteksi URL koneksi basis data otomatis dan memberikan fallback ke lingkungan lokal:

Bukti Implementasi Kode Sumber:
backend/src/config/database.js 
const mysql = require('mysql2/promise');
require('dotenv').config();

// Konfigurasi Dinamis: Cloud Database URL (Railway) vs Localhost
const poolConfig = process.env.MYSQL_URL || process.env.DATABASE_URL
  ? {
      uri: process.env.MYSQL_URL || process.env.DATABASE_URL,
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0,
      timezone: '+07:00',
      charset: 'utf8mb4'
    }
  : {
      host: process.env.DB_HOST || 'localhost',
      port: parseInt(process.env.DB_PORT || '3306'),
      user: process.env.DB_USER || 'root',
      password: process.env.DB_PASSWORD || '',
      database: process.env.DB_NAME || 'isa_career_hrd',
      waitForConnections: true,
      connectionLimit: 10,
      queueLimit: 0,
      timezone: '+07:00',
      charset: 'utf8mb4'
    };

const pool = mysql.createPool(poolConfig);

pool.getConnection()
  .then(conn => {
    console.log('[DB] MySQL Connected — Pool Ready');
    conn.release();
  })
  .catch(err => console.error('[DB] Connection Error:', err.message));

module.exports = pool;

FROM products 
WHERE status = 1 
ORDER BY sort_order ASC, id ASC;
