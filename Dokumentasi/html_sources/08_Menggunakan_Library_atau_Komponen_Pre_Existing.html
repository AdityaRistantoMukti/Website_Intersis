8.3. Konfigurasi Inisialisasi Modul Multi-Bahasa i18next

Lokasi Berkas: frontend/src/lib/i18n.js

Konfigurasi terpusat pustaka i18next untuk menyinkronkan kamus bahasa Indonesia dan Inggris:

Bukti Implementasi Kode Sumber (Visual Studio Code):
frontend/src/lib/i18n.js
VS CODE • UTF-8
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import LanguageDetector from 'i18next-browser-languagedetector';

import idCommon from '../locales/id/common.json';
import idHome from '../locales/id/home.json';
import idProducts from '../locales/id/products.json';
import enCommon from '../locales/en/common.json';
import enHome from '../locales/en/home.json';
import enProducts from '../locales/en/products.json';

i18n
  .use(LanguageDetector)
  .use(initReactI18next)
  .init({
    resources: {
      id: { common: idCommon, home: idHome, products: idProducts },
      en: { common: enCommon, home: enHome, products: enProducts }
    },
    fallbackLng: 'id',
    defaultNS: 'common',
    interpolation: { escapeValue: false }
  });

export default i18n;
